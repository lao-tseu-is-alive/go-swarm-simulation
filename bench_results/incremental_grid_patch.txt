+// updateActorInGrid moves an actor to its new cell if it changed.
+// Uses swap-and-truncate for O(1) removal (order doesn't matter in cells).
+func (w *WorldActor) updateActorInGrid(actor *Entity, cellSize float64) {
+	gx, gy := int(actor.Pos.X/cellSize), int(actor.Pos.Y/cellSize)
+	newCell := gridKey{x: gx, y: gy}
+	oldCell, exists := w.actorCells[actor.ID]
+
+	if exists && oldCell == newCell {
+		return // Actor stayed in same cell - most common case
+	}
+
+	// Remove from old cell if it exists
+	if exists {
+		w.removeFromCell(oldCell, actor.ID)
+	}
+
+	// Add to new cell
+	w.grid[newCell] = append(w.grid[newCell], actor)
+	w.actorCells[actor.ID] = newCell
+}

+// removeFromCell removes an actor from a specific grid cell.
+// Uses swap-with-last for O(1) unordered removal.
+func (w *WorldActor) removeFromCell(cell gridKey, actorID string) {
